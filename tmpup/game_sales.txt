function styleHead_(s,n){ s.getRange(1,1,1,n).setFontWeight('bold').setBackground('#26324a').setFontColor('#ffffff'); }
function nameRanges_(ss,s){

  names.forEach(function(nm,i){ var old=ss.getRangeByName(nm); if(old) ss.removeNamedRange(nm); ss.setNamedRange(nm, s.getRange(2+i,2)); });

/* ---------- 1行足す（手数料は数式で計算） ---------- */
function appendSale_(row){
  var s = SpreadsheetApp.getActive().getSheetByName(SH.SALES);
  var r = s.getLastRow()+1;
  s.getRange(r,1,1,8).setValues([[row.date,row.site,row.title,row.order,row.qty,row.amount,row.cur,row.cur==='JPY'?1:'']]);
  if(row.cur!=='JPY') s.getRange(r,8).setFormula('=RATE_FX');
  s.getRange(r,9).setFormula('=ROUND(F'+r+'*H'+r+',0)');
  if(row.site==='BOOTH'){
    s.getRange(r,10).setFormula('=ROUND(I'+r+'*BOOTH_RATE+BOOTH_FIX,0)');
    s.getRange(r,11).setValue(0); s.getRange(r,12).setValue(0); s.getRange(r,13).setValue(0);
  } else {
    s.getRange(r,10).setFormula('=ROUND(I'+r+'*ITCH_SHARE,0)');
    s.getRange(r,11).setFormula('=ROUND((F'+r+'*PROC_RATE+PROC_FIX*E'+r+')*H'+r+',0)');
    s.getRange(r,13).setFormula('=ROUND(I'+r+'*ITCH_WHT,0)');
    s.getRange(r,12).setFormula('=ROUND((I'+r+'-J'+r+'-K'+r+'-M'+r+')*PAYONEER,0)');
  }
  s.getRange(r,14).setFormula('=I'+r+'-J'+r+'-K'+r+'-L'+r+'-M'+r);
  s.getRange(r,15,1,3).setValues([[row.src,row.check||'',row.memo||'']]);
  return r;
}

/* ---------- BOOTH：購入メールから ---------- */
function importBooth_(){
  var label = GmailApp.getUserLabelByName('ゲーム販売_取込済み') || GmailApp.createLabel('ゲーム販売_取込済み');
  var start = SpreadsheetApp.getActive().getRangeByName('START_DATE').getValue();
  var after = Utilities.formatDate(new Date(start),'Asia/Tokyo','yyyy/MM/dd');
  var q = 'from:(booth.pm OR pixiv.net) -label:ゲーム販売_取込済み after:'+after+' (購入 OR 注文 OR 売れ)';
  var threads = GmailApp.search(q, 0, 50);
  var games = SpreadsheetApp.getActive().getSheetByName(SH.GAMES).getDataRange().getValues().slice(1);
  var done = existingOrders_();
  var n = 0;
  threads.forEach(function(th){
    th.getMessages().forEach(function(msg){
      var body = msg.getPlainBody(), subj = msg.getSubject();
      if(!/購入|注文/.test(subj+body)) return;
      if(/発送|メッセージ|お問い合わせ|ログイン|パスワード/.test(subj) && !/購入/.test(subj)) return;
      var order = ((body.match(/注文番号[\s:：#]*([0-9]{5,})/)||[])[1]) || '';
      if(order && done['BOOTH:'+order]) return;
      var hits = games.filter(function(g){ return g[0] && (body.indexOf(g[0])>=0 || body.indexOf(String(g[2]))>=0); });
      var title = hits.length===1 ? hits[0][0] : (hits.length>1 ? hits.map(function(h){return h[0]}).join('＋') : '');
      var yen = (body.match(/(?:合計|小計|金額|価格)[^\n¥￥0-9]{0,15}[¥￥]\s?([0-9,]+)/)||body.match(/[¥￥]\s?([0-9,]+)/)||[])[1];
      var amount = yen ? Number(yen.replace(/,/g,'')) : '';
      var ok = order && title && amount!=='' && hits.length===1;
      appendSale_({date:msg.getDate(), site:'BOOTH', title:title||'（不明）', order:order||'（不明）', qty:1, amount:amount===''?0:amount, cur:'JPY',
        src:'BOOTH メール：'+subj, check: ok?'':'要確認（メールの読み取りに不明な所あり）'});
      if(order) done['BOOTH:'+order]=true; n++;
    });
    th.addLabel(label);
  });
  return n;
}
function existingOrders_(){
  var s = SpreadsheetApp.getActive().getSheetByName(SH.SALES); var v = s.getDataRange().getValues(); var o={};
  for(var i=1;i<v.length;i++) if(v[i][3]) o[v[i][1]+':'+v[i][3]]=true; return o;
}

/* ---------- itch.io：API の合計の差分から ---------- */
function importItch_(){
  var key = PropertiesService.getScriptProperties().getProperty('ITCH_API_KEY');
  if(!key) return '鍵なし';
  var res = UrlFetchApp.fetch('https://api.itch.io/profile/games',{headers:{Authorization:'Bearer '+key},muteHttpExceptions:true});
  if(res.getResponseCode()!==200) return 'エラー '+res.getResponseCode();
  var data = JSON.parse(res.getContentText()); var games = data.games||[];
  var ss = SpreadsheetApp.getActive(), it = ss.getSheetByName(SH.ITCH);
  var gl = ss.getSheetByName(SH.GAMES).getDataRange().getValues().slice(1);
  var last = {}; var hv = it.getDataRange().getValues();
  for(var i=1;i<hv.length;i++) last[String(hv[i][1])] = {cnt:Number(hv[i][3]), amt:Number(hv[i][4])};
  var n=0, now=new Date();
  games.forEach(function(g){
    var id=String(g.id), cnt=Number(g.purchases_count||0), amt=0, cur='USD';
    (g.earnings||[]).forEach(function(e){ if(e.currency==='USD'){ amt+=Number(e.amount||0); cur='USD'; } });
    var name = (gl.filter(function(x){return String(x[4])===id})[0]||[])[0] || g.title;
    var prev = last[id];
    if(prev && cnt>prev.cnt){
      var dq=cnt-prev.cnt, da=(amt-prev.amt)/100;
      appendSale_({date:now, site:'itch.io', title:name, order:'itch '+id+' #'+prev.cnt+'→'+cnt, qty:dq, amount:da, cur:'USD',
        src:'itch.io API（累計の差分）', check: da>0 ? '' : '要確認（金額の差が0以下）'});
      n+=dq;
    }
    if(!prev || cnt!==prev.cnt || amt!==prev.amt) it.appendRow([now,id,name,cnt,amt,cur]);
  });
  return n;
}

function importAll(){
  var lock = LockService.getScriptLock(); if(!lock.tryLock(30000)) return;
  var b='', i='', memo='';
  try{ b = importBooth_(); }catch(e){ b='エラー'; memo+='BOOTH: '+e.message+' '; }
  try{ i = importItch_(); }catch(e){ i='エラー'; memo+='itch: '+e.message; }
  SpreadsheetApp.getActive().getSheetByName(SH.LOG).appendRow([new Date(), b, i, memo]);
  lock.releaseLock();
}

function setItchKey(){
  var ui = SpreadsheetApp.getUi();
  var r = ui.prompt('itch.io の API キー','itch.io → 設定 → APIキー で作った鍵を貼ってください（シートには表示されません）',ui.ButtonSet.OK_CANCEL);
  if(r.getSelectedButton()===ui.Button.OK && r.getResponseText().trim()){
    PropertiesService.getScriptProperties().setProperty('ITCH_API_KEY', r.getResponseText().trim());
    ui.alert('保存しました。次の取り込みから itch.io の販売も入ります。');
  }
}

function installTriggers(){
  ScriptApp.getProjectTriggers().forEach(function(t){ if(t.getHandlerFunction()==='importAll') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('importAll').timeBased().everyHours(1).create();
  return 'trigger ok';
}
