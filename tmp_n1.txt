/* 空いている最初の行（A〜E が空）に書く */
function noteAppend_(row){
  var s = SpreadsheetApp.getActive().getSheetByName(NS.SALES);
  var v = s.getRange(2,1,N_ROWS,5).getValues(), r=-1;
  for(var i=0;i<v.length;i++){ if(v[i].join('')===''){ r=i+2; break; } }
  if(r<0) throw new Error('販売明細の行が足りません');
  s.getRange(r,1,1,5).setValues([[row.date,row.kind,row.title,row.price,row.pay]]);
  s.getRange(r,9,1,3).setValues([[row.src||'',row.check||'',row.memo||'']]);
  return r;
}

function importNote(){
  var lock = LockService.getScriptLock(); if(!lock.tryLock(30000)) return;
  var n=0, memo='';
  try{
    var label = GmailApp.getUserLabelByName('note販売_取込済み') || GmailApp.createLabel('note販売_取込済み');
    var start = SpreadsheetApp.getActive().getRangeByName('N_START').getValue();
    var after = Utilities.formatDate(new Date(start),'Asia/Tokyo','yyyy/MM/dd');
    var th = GmailApp.search('from:note.com -label:note販売_取込済み after:'+after+' (購入 OR 売れ OR サポート OR メンバー)',0,50);
    th.forEach(function(t){
      t.getMessages().forEach(function(msg){
        var subj=msg.getSubject(), body=msg.getPlainBody();
        if(!/購入|売れ|サポート|メンバーシップに参加/.test(subj)) return;
        var title=((body.match(/[「『]([^」』]{2,120})[」』]/)||subj.match(/[「『]([^」』]{2,120})[」』]/)||[])[1])||'';
        var yen=(body.match(/([0-9,]+)\s*円/)||body.match(/[¥￥]\s?([0-9,]+)/)||[])[1];
        var price=yen?Number(yen.replace(/,/g,'')):'';
        var kind=/定期購読/.test(subj+body)?'定期購読マガジン':/マガジン/.test(subj)?'有料マガジン':/メンバーシップ/.test(subj+body)?'メンバーシップ':/サポート|チップ/.test(subj)?'チップ':'有料記事';
        noteAppend_({date:msg.getDate(), kind:kind, title:title||'（不明）', price:price===''?'':price, pay:'クレジットカード',
          src:'note メール：'+subj, check:'要確認（決済方法・金額を note の売上管理で確かめる）'});
        n++;
      });
      t.addLabel(label);
    });
  }catch(e){ memo=e.message; }
  SpreadsheetApp.getActive().getSheetByName(NS.LOG).appendRow([new Date(), n, memo]);
  lock.releaseLock();
}

function installNoteTrigger(){
  ScriptApp.getProjectTriggers().forEach(function(t){ if(t.getHandlerFunction()==='importNote') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('importNote').timeBased().everyHours(1).create();
  return 'trigger ok';
}

  var mrows=[];
  for(var m=0;m<12;m++){
    var inM=function(r){ return new Date(r[0]).getMonth()===m; };
    var s1=sales.filter(inM), p1=pays.filter(inM), c1=costs.filter(inM);
    var u=sum(s1,3), f=sum(s1,5)+sum(s1,6)+sum(p1,2), c=sum(c1,3);

  ]});

/* ===== 税理士用データの書き出し（共通） v1 2026-11-21 Sanuki Trading
   eBay の「税理士用データを書き出す（社長の管理表と同じ形）」と同じ色・同じ並び（①はじめに → ②確定申告のまとめ → ③月別推移 → ④以降 明細）。
   新しいファイルを作り、Excel（.xlsx）と PDF のダウンロードの道をダイアログに出す。 */
var TX = { TITLE:'#1A3C5E', HEAD:'#2E4057', NOTE:'#FFF9E6', SEP:'#D9E2F3', TOTAL:'#FFE699', SUB:'#EDF1F9', ALT:'#F7F9FC', OUT:'#FDEBD3' };
var KAMOKU = ['消耗品費','通信費','支払手数料','広告宣伝費','外注費','支払報酬','諸会費','新聞図書費','荷造包装費','旅費交通費','雑費'];

function txYear_(){
  var ui = SpreadsheetApp.getUi(); var y = new Date().getFullYear();
  var r = ui.prompt('税理士用データを書き出す','何年の分ですか（西暦）。空のままOKで '+y+' 年',ui.ButtonSet.OK_CANCEL);
  if(r.getSelectedButton()!==ui.Button.OK) return null;
  var t = r.getResponseText().trim(); return t ? Number(t) : y;
}
function txInYear_(d,y){ d = d instanceof Date ? d : new Date(d); return !isNaN(d) && d.getFullYear()===y; }
function txNum_(v){ var n = Number(v); return isNaN(n) ? 0 : n; }

/* 科目の列（経費シートの F 列）が無ければ足す */
function txEnsureKamoku_(sheetName, col){
  var s = SpreadsheetApp.getActive().getSheetByName(sheetName); if(!s) return;
  if(String(s.getRange(1,col).getValue())!=='科目（税理士の区分）'){
    s.getRange(1,col).setValue('科目（税理士の区分）').setFontWeight('bold').setBackground('#26324a').setFontColor('#ffffff');
  }
  var rule = SpreadsheetApp.newDataValidation().requireValueInList(KAMOKU,true).setAllowInvalid(true).build();
  s.getRange(2,col,999,1).setDataValidation(rule);
}

/* book = { name, intro:[[文]], summary:[[項目,金額,種類('h'見出し/'s'小計/'t'合計/'n'注意/''),説明]], sheets:[{name, head, rows, money:[列番号1始まり], date:[列], totalLabelCol, sumCols:[列]}] } */
function txWriteBook_(book){
  var ss = SpreadsheetApp.create(book.name);
  var first = ss.getSheets()[0];

  // ① はじめに
  var a = first.setName('①はじめに');
  a.getRange(1,1).setValue(book.name).setFontSize(14).setFontWeight('bold').setBackground(TX.TITLE).setFontColor('#ffffff');
  a.getRange(1,1,1,2).setBackground(TX.TITLE);
  for(var i=0;i<book.intro.length;i++){
    var c = a.getRange(3+i,1).setValue(book.intro[i][0]).setWrap(true);
    if(book.intro[i][1]==='h') a.getRange(3+i,1,1,2).setBackground(TX.HEAD).setFontColor('#ffffff').setFontWeight('bold');
    if(book.intro[i][1]==='n') a.getRange(3+i,1,1,2).setBackground(TX.NOTE);
  }
  a.setColumnWidth(1,900); a.setColumnWidth(2,40);

  // ② 確定申告のまとめ
  var b = ss.insertSheet('②確定申告のまとめ');
  b.getRange(1,1,1,3).merge().setValue(book.name+'　確定申告のまとめ').setFontWeight('bold').setFontSize(13).setBackground(TX.TITLE).setFontColor('#ffffff');
  b.getRange(2,1,1,3).setValues([['項目','金額（円）','説明']]).setBackground(TX.HEAD).setFontColor('#ffffff').setFontWeight('bold');
  for(var j=0;j<book.summary.length;j++){
    var row = book.summary[j], r = 3+j;
    b.getRange(r,1,1,3).setValues([[row[0], row[1]===''?'':row[1], row[3]||'']]);
    var kind=row[2]||'';
    if(kind==='h') b.getRange(r,1,1,3).setBackground(TX.SEP).setFontWeight('bold');
    else if(kind==='s') b.getRange(r,1,1,3).setBackground(TX.SUB).setFontWeight('bold');
    else if(kind==='t') b.getRange(r,1,1,3).setBackground(TX.TOTAL).setFontWeight('bold');
    else if(kind==='n') b.getRange(r,1,1,3).setBackground(TX.NOTE);
  }
  b.getRange(3,2,Math.max(1,book.summary.length),1).setNumberFormat('#,##0');
  b.setColumnWidth(1,300); b.setColumnWidth(2,130); b.setColumnWidth(3,560); b.setFrozenRows(2);

  // ③ 以降の表
  book.sheets.forEach(function(t){
    var s = ss.insertSheet(t.name);
    var n = t.head.length;
    s.getRange(1,1,1,n).merge().setValue(t.title||t.name).setFontWeight('bold').setBackground(TX.TITLE).setFontColor('#ffffff');
    s.getRange(2,1,1,n).setValues([t.head]).setBackground(TX.HEAD).setFontColor('#ffffff').setFontWeight('bold').setWrap(true);
    var rows = t.rows.slice();
    if(rows.length){
      s.getRange(3,1,rows.length,n).setValues(rows);
      for(var k=0;k<rows.length;k++) if(k%2===1) s.getRange(3+k,1,1,n).setBackground(TX.ALT);
      if(t.outRows) t.outRows.forEach(function(ix){ s.getRange(3+ix,1,1,n).setBackground(TX.OUT); });
    }
    var tr = 3+rows.length;
    if(t.sumCols && t.sumCols.length){
      s.getRange(tr,1).setValue('合計');
      t.sumCols.forEach(function(cn){
        var tot=0; rows.forEach(function(x){ tot+=txNum_(x[cn-1]); });
        s.getRange(tr,cn).setValue(tot);
      });
      s.getRange(tr,1,1,n).setBackground(TX.TOTAL).setFontWeight('bold');
    }
    (t.money||[]).forEach(function(cn){ s.getRange(3,cn,rows.length+1,1).setNumberFormat('#,##0'); });
    (t.date||[]).forEach(function(cn){ if(rows.length) s.getRange(3,cn,rows.length,1).setNumberFormat(t.dateFmt||'yyyy/mm/dd'); });
    if(t.note){ s.getRange(tr+2,1,1,n).merge().setValue(t.note).setBackground(TX.NOTE).setWrap(true); }
    s.setFrozenRows(2);
    for(var w=1; w<=n; w++) s.setColumnWidth(w, (t.widths && t.widths[w-1]) || 110);
  });

  SpreadsheetApp.flush();
  var id = ss.getId();
  return { id:id, url:ss.getUrl(),
    xlsx:'https://docs.google.com/spreadsheets/d/'+id+'/export?format=xlsx',
    pdf:'https://docs.google.com/spreadsheets/d/'+id+'/export?format=pdf&portrait=false&fitw=true&gridlines=false' };
}
