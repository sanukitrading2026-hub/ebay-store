/* ---------- 1行足す（手数料は数式で計算） ---------- */
function appendSale_(row){
  var s = SpreadsheetApp.getActive().getSheetByName(SH.SALES);
  var r = s.getLastRow()+1;
  s.getRange(r,1,1,8).setValues([[row.date,row.site,row.title,row.order,row.qty,row.amount,row.cur,row.cur==='JPY'?1:'']]);
  if(row.cur!=='JPY') s.getRange(r,8).setFormula('=RATE_FX');
  s.getRange(r,9).setFormula('=ROUND(F'+r+'*H'+r+',0)');
  if(row.site==='BOOTH'){
    s.getRange(r,10).setFormula('=ROUND(I'+r+'*BOOTH_RATE+BOOTH_FIX,0)');
    s.getRange(r,11).setValue(0); s.getRange(r,12).setValue(0); s.getRange(r,13).setValue(0);
  } else {
    s.getRange(r,10).setFormula('=ROUND(I'+r+'*ITCH_SHARE,0)');
    s.getRange(r,11).setFormula('=ROUND((F'+r+'*PROC_RATE+PROC_FIX*E'+r+')*H'+r+',0)');
    s.getRange(r,13).setFormula('=ROUND(I'+r+'*ITCH_WHT,0)');
    s.getRange(r,12).setFormula('=ROUND((I'+r+'-J'+r+'-K'+r+'-M'+r+')*PAYONEER,0)');
  }
  s.getRange(r,14).setFormula('=I'+r+'-J'+r+'-K'+r+'-L'+r+'-M'+r);
  s.getRange(r,15,1,3).setValues([[row.src,row.check||'',row.memo||'']]);
  return r;
}

/* ---------- BOOTH：購入メールから ---------- */
function importBooth_(){
  var label = GmailApp.getUserLabelByName('ゲーム販売_取込済み') || GmailApp.createLabel('ゲーム販売_取込済み');
  var start = SpreadsheetApp.getActive().getRangeByName('START_DATE').getValue();
  var after = Utilities.formatDate(new Date(start),'Asia/Tokyo','yyyy/MM/dd');
  var q = 'from:(booth.pm OR pixiv.net) -label:ゲーム販売_取込済み after:'+after+' (購入 OR 注文 OR 売れ)';
  var threads = GmailApp.search(q, 0, 50);
  var games = SpreadsheetApp.getActive().getSheetByName(SH.GAMES).getDataRange().getValues().slice(1);
  var done = existingOrders_();
  var n = 0;
  threads.forEach(function(th){
    th.getMessages().forEach(function(msg){
      var body = msg.getPlainBody(), subj = msg.getSubject();
      if(!/購入|注文/.test(subj+body)) return;
      if(/発送|メッセージ|お問い合わせ|ログイン|パスワード/.test(subj) && !/購入/.test(subj)) return;
      var order = ((body.match(/注文番号[\s:：#]*([0-9]{5,})/)||[])[1]) || '';
      if(order && done['BOOTH:'+order]) return;
      var hits = games.filter(function(g){ return g[0] && (body.indexOf(g[0])>=0 || body.indexOf(String(g[2]))>=0); });
      var title = hits.length===1 ? hits[0][0] : (hits.length>1 ? hits.map(function(h){return h[0]}).join('＋') : '');
      var yen = (body.match(/(?:合計|小計|金額|価格)[^\n¥￥0-9]{0,15}[¥￥]\s?([0-9,]+)/)||body.match(/[¥￥]\s?([0-9,]+)/)||[])[1];
      var amount = yen ? Number(yen.replace(/,/g,'')) : '';
      var ok = order && title && amount!=='' && hits.length===1;
      appendSale_({date:msg.getDate(), site:'BOOTH', title:title||'（不明）', order:order||'（不明）', qty:1, amount:amount===''?0:amount, cur:'JPY',
        src:'BOOTH メール：'+subj, check: ok?'':'要確認（メールの読み取りに不明な所あり）'});
      if(order) done['BOOTH:'+order]=true; n++;
    });
    th.addLabel(label);
  });
  return n;
}
function existingOrders_(){
  var s = SpreadsheetApp.getActive().getSheetByName(SH.SALES); var v = s.getDataRange().getValues(); var o={};
  for(var i=1;i<v.length;i++) if(v[i][3]) o[v[i][1]+':'+v[i][3]]=true; return o;
}

/* ---------- itch.io：API の合計の差分から ---------- */
function importItch_(){
  var key = PropertiesService.getScriptProperties().getProperty('ITCH_API_KEY');
  if(!key) return '鍵なし';
  var res = UrlFetchApp.fetch('https://api.itch.io/profile/games',{headers:{Authorization:'Bearer '+key},muteHttpExceptions:true});
  if(res.getResponseCode()!==200) return 'エラー '+res.getResponseCode();
  var data = JSON.parse(res.getContentText()); var games = data.games||[];
  var ss = SpreadsheetApp.getActive(), it = ss.getSheetByName(SH.ITCH);
  var gl = ss.getSheetByName(SH.GAMES).getDataRange().getValues().slice(1);
  var last = {}; var hv = it.getDataRange().getValues();
  for(var i=1;i<hv.length;i++) last[String(hv[i][1])] = {cnt:Number(hv[i][3]), amt:Number(hv[i][4])};
  var n=0, now=new Date();
  games.forEach(function(g){
    var id=String(g.id), cnt=Number(g.purchases_count||0), amt=0, cur='USD';
    (g.earnings||[]).forEach(function(e){ if(e.currency==='USD'){ amt+=Number(e.amount||0); cur='USD'; } });
    var name = (gl.filter(function(x){return String(x[4])===id})[0]||[])[0] || g.title;
    var prev = last[id];
    if(prev && cnt>prev.cnt){
      var dq=cnt-prev.cnt, da=(amt-prev.amt)/100;
      appendSale_({date:now, site:'itch.io', title:name, order:'itch '+id+' #'+prev.cnt+'→'+cnt, qty:dq, amount:da, cur:'USD',
        src:'itch.io API（累計の差分）', check: da>0 ? '' : '要確認（金額の差が0以下）'});
      n+=dq;
    }
    if(!prev || cnt!==prev.cnt || amt!==prev.amt) it.appendRow([now,id,name,cnt,amt,cur]);
  });
  return n;
}

function importAll(){
  var lock = LockService.getScriptLock(); if(!lock.tryLock(30000)) return;
  var b='', i='', memo='';
  try{ b = importBooth_(); }catch(e){ b='エラー'; memo+='BOOTH: '+e.message+' '; }
  try{ i = importItch_(); }catch(e){ i='エラー'; memo+='itch: '+e.message; }
  SpreadsheetApp.getActive().getSheetByName(SH.LOG).appendRow([new Date(), b, i, memo]);
  lock.releaseLock();
}

function setItchKey(){
  var ui = SpreadsheetApp.getUi();
  var r = ui.prompt('itch.io の API キー','itch.io → 設定 → APIキー で作った鍵を貼ってください（シートには表示されません）',ui.ButtonSet.OK_CANCEL);
  if(r.getSelectedButton()===ui.Button.OK && r.getResponseText().trim()){
    PropertiesService.getScriptProperties().setProperty('ITCH_API_KEY', r.getResponseText().trim());
    ui.alert('保存しました。次の取り込みから itch.io の販売も入ります。');
  }
}

function installTriggers(){
  ScriptApp.getProjectTriggers().forEach(function(t){ if(t.getHandlerFunction()==='importAll') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('importAll').timeBased().everyHours(1).create();
  return 'trigger ok';
}

   eBay の「税理士用データを書き出す（社長の管理表と同じ形）」と同じ色・同じ並び（①はじめに → ②確定申告のまとめ → ③月別推移 → ④以降 明細）。
   新しいファイルを作り、Excel（.xlsx）と PDF のダウンロードの道をダイアログに出す。 */
var TX = { TITLE:'#1A3C5E', HEAD:'#2E4057', NOTE:'#FFF9E6', SEP:'#D9E2F3', TOTAL:'#FFE699', SUB:'#EDF1F9', ALT:'#F7F9FC', OUT:'#FDEBD3' };
var KAMOKU = ['消耗品費','通信費','支払手数料','広告宣伝費','外注費','支払報酬','諸会費','新聞図書費','荷造包装費','旅費交通費','雑費'];

function txYear_(){
  var ui = SpreadsheetApp.getUi(); var y = new Date().getFullYear();
  var r = ui.prompt('税理士用データを書き出す','何年の分ですか（西暦）。空のままOKで '+y+' 年',ui.ButtonSet.OK_CANCEL);
  if(r.getSelectedButton()!==ui.Button.OK) return null;
  var t = r.getResponseText().trim(); return t ? Number(t) : y;
}
function txInYear_(d,y){ d = d instanceof Date ? d : new Date(d); return !isNaN(d) && d.getFullYear()===y; }
function txNum_(v){ var n = Number(v); return isNaN(n) ? 0 : n; }

/* 科目の列（経費シートの F 列）が無ければ足す */
function txEnsureKamoku_(sheetName, col){
  var s = SpreadsheetApp.getActive().getSheetByName(sheetName); if(!s) return;
  if(String(s.getRange(1,col).getValue())!=='科目（税理士の区分）'){
    s.getRange(1,col).setValue('科目（税理士の区分）').setFontWeight('bold').setBackground('#26324a').setFontColor('#ffffff');
  }
  var rule = SpreadsheetApp.newDataValidation().requireValueInList(KAMOKU,true).setAllowInvalid(true).build();
  s.getRange(2,col,999,1).setDataValidation(rule);
}

/* book = { name, intro:[[文]], summary:[[項目,金額,種類('h'見出し/'s'小計/'t'合計/'n'注意/''),説明]], sheets:[{name, head, rows, money:[列番号1始まり], date:[列], totalLabelCol, sumCols:[列]}] } */
function txWriteBook_(book){
  var ss = SpreadsheetApp.create(book.name);
  var first = ss.getSheets()[0];

  // ① はじめに
  var a = first.setName('①はじめに');
  a.getRange(1,1).setValue(book.name).setFontSize(14).setFontWeight('bold').setBackground(TX.TITLE).setFontColor('#ffffff');
  a.getRange(1,1,1,2).setBackground(TX.TITLE);
  for(var i=0;i<book.intro.length;i++){
    var c = a.getRange(3+i,1).setValue(book.intro[i][0]).setWrap(true);
    if(book.intro[i][1]==='h') a.getRange(3+i,1,1,2).setBackground(TX.HEAD).setFontColor('#ffffff').setFontWeight('bold');
    if(book.intro[i][1]==='n') a.getRange(3+i,1,1,2).setBackground(TX.NOTE);
  }
  a.setColumnWidth(1,900); a.setColumnWidth(2,40);

  // ② 確定申告のまとめ
  var b = ss.insertSheet('②確定申告のまとめ');
  b.getRange(1,1,1,3).merge().setValue(book.name+'　確定申告のまとめ').setFontWeight('bold').setFontSize(13).setBackground(TX.TITLE).setFontColor('#ffffff');
  b.getRange(2,1,1,3).setValues([['項目','金額（円）','説明']]).setBackground(TX.HEAD).setFontColor('#ffffff').setFontWeight('bold');
  for(var j=0;j<book.summary.length;j++){
    var row = book.summary[j], r = 3+j;
    b.getRange(r,1,1,3).setValues([[row[0], row[1]===''?'':row[1], row[3]||'']]);
    var kind=row[2]||'';
    if(kind==='h') b.getRange(r,1,1,3).setBackground(TX.SEP).setFontWeight('bold');
    else if(kind==='s') b.getRange(r,1,1,3).setBackground(TX.SUB).setFontWeight('bold');
    else if(kind==='t') b.getRange(r,1,1,3).setBackground(TX.TOTAL).setFontWeight('bold');
    else if(kind==='n') b.getRange(r,1,1,3).setBackground(TX.NOTE);
  }
  b.getRange(3,2,Math.max(1,book.summary.length),1).setNumberFormat('#,##0');
  b.setColumnWidth(1,300); b.setColumnWidth(2,130); b.setColumnWidth(3,560); b.setFrozenRows(2);

  // ③ 以降の表
  book.sheets.forEach(function(t){
    var s = ss.insertSheet(t.name);
    var n = t.head.length;
    s.getRange(1,1,1,n).merge().setValue(t.title||t.name).setFontWeight('bold').setBackground(TX.TITLE).setFontColor('#ffffff');
    s.getRange(2,1,1,n).setValues([t.head]).setBackground(TX.HEAD).setFontColor('#ffffff').setFontWeight('bold').setWrap(true);
    var rows = t.rows.slice();
    if(rows.length){
      s.getRange(3,1,rows.length,n).setValues(rows);
      for(var k=0;k<rows.length;k++) if(k%2===1) s.getRange(3+k,1,1,n).setBackground(TX.ALT);
      if(t.outRows) t.outRows.forEach(function(ix){ s.getRange(3+ix,1,1,n).setBackground(TX.OUT); });
    }
    var tr = 3+rows.length;
    if(t.sumCols && t.sumCols.length){
      s.getRange(tr,1).setValue('合計');
      t.sumCols.forEach(function(cn){
        var tot=0; rows.forEach(function(x){ tot+=txNum_(x[cn-1]); });
        s.getRange(tr,cn).setValue(tot);
      });
      s.getRange(tr,1,1,n).setBackground(TX.TOTAL).setFontWeight('bold');
    }
    (t.money||[]).forEach(function(cn){ s.getRange(3,cn,rows.length+1,1).setNumberFormat('#,##0'); });
    (t.date||[]).forEach(function(cn){ if(rows.length) s.getRange(3,cn,rows.length,1).setNumberFormat(t.dateFmt||'yyyy/mm/dd'); });
    if(t.note){ s.getRange(tr+2,1,1,n).merge().setValue(t.note).setBackground(TX.NOTE).setWrap(true); }
    s.setFrozenRows(2);
    for(var w=1; w<=n; w++) s.setColumnWidth(w, (t.widths && t.widths[w-1]) || 110);
  });

  SpreadsheetApp.flush();
  var id = ss.getId();
  return { id:id, url:ss.getUrl(),
    xlsx:'https://docs.google.com/spreadsheets/d/'+id+'/export?format=xlsx',
    pdf:'https://docs.google.com/spreadsheets/d/'+id+'/export?format=pdf&portrait=false&fitw=true&gridlines=false' };
}

  var db = SpreadsheetApp.getActive().getSheetByName(dashName);

}
